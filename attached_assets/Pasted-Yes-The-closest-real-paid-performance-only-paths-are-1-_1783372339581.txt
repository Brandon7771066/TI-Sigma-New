schema.py

from pydantic import BaseModel
from typing import Literal
TruthLabel = Literal["TRUE", "FALSE", "INDETERMINATE", "META_INDETERMINATE", "N_A"]
class TruthEvaluation(BaseModel):
    label: TruthLabel
    pd_degree: float          # real truth degree: 0=false, 0.5=indeterminate, 1=true
    pd_modality: float        # imaginary / weirdness / category-error loading
    tau_delta: float          # gap between true-as-stated and instantiated
    authority_loading: float  # dependence on source trust
    confidence: float
    explanation: str

truth_engine.py

from schema import TruthEvaluation
def clamp(x: float) -> float:
    return max(0.0, min(1.0, x))
def evaluate_statement(statement: str) -> TruthEvaluation:
    s = statement.lower().strip()
    # 1. N/A screen
    na_markers = [
        "will happen", "in the future", "unknown", "not enough information",
        "i haven't decided", "cannot determine", "unavailable data"
    ]
    if any(m in s for m in na_markers):
        return TruthEvaluation(
            label="N_A",
            pd_degree=0.5,
            pd_modality=0.9,
            tau_delta=0.8,
            authority_loading=0.2,
            confidence=0.72,
            explanation="N/A screen triggered: current answer is not available from the given information."
        )
    # 2. Meta-Indeterminate screen
    mi_markers = [
        "this sentence is false",
        "number 7 is jealous",
        "square circle",
        "same now as 20 years ago",
        "both true and false"
    ]
    if any(m in s for m in mi_markers):
        return TruthEvaluation(
            label="META_INDETERMINATE",
            pd_degree=0.5,
            pd_modality=0.95,
            tau_delta=0.9,
            authority_loading=0.05,
            confidence=0.85,
            explanation="MI screen triggered: proposition appears self-defeating, category-confused, or coherence-breaking."
        )
    # 3. crude truth-spectrum placeholder
    false_markers = ["sydney is the capital of australia", "2+2=5", "earth is flat"]
    true_markers = ["canberra is the capital of australia", "2+2=4", "water contains hydrogen"]
    if any(m in s for m in true_markers):
        label = "TRUE"
        pd = 0.95
        explanation = "High real-axis truth degree based on direct factual match."
    elif any(m in s for m in false_markers):
        label = "FALSE"
        pd = 0.05
        explanation = "Low real-axis truth degree based on direct factual contradiction."
    else:
        label = "INDETERMINATE"
        pd = 0.52
        explanation = "No decisive truth/falsity marker; classified as stable middle-region indeterminacy."
    return TruthEvaluation(
        label=label,
        pd_degree=pd,
        pd_modality=0.25 if label in ["TRUE", "FALSE"] else 0.55,
        tau_delta=0.3,
        authority_loading=0.4,
        confidence=0.7,
        explanation=explanation
    )

main.py

import streamlit as st
import pandas as pd
from truth_engine import evaluate_statement
st.set_page_config(page_title="TI Sigma Truth Evaluator", layout="wide")
st.title("TI Sigma Truth Evaluator")
st.caption("Five-valued truth labels + four truth axes for AI output evaluation.")
statement = st.text_area("Paste an AI output or claim:", height=180)
if st.button("Evaluate"):
    result = evaluate_statement(statement)
    st.subheader("Truth Label")
    st.write(result.label)
    st.subheader("Truth Axes")
    df = pd.DataFrame([{
        "PD Degree": result.pd_degree,
        "PD Modality": result.pd_modality,
        "τ/δ Separability": result.tau_delta,
        "Authority Loading": result.authority_loading,
        "Confidence": result.confidence,
    }])
    st.dataframe(df, use_container_width=True)
    st.subheader("Explanation")
    st.write(result.explanation)
    st.subheader("JSON")
    st.json(result.model_dump())
