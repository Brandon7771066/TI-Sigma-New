## Missing `schema.py`

Use this:

```python
from pydantic import BaseModel, Field
from typing import Literal, Dict, List

TruthLabel = Literal[
    "TRUE",
    "FALSE",
    "INDETERMINATE",
    "META_INDETERMINATE",
    "N_A",
    "NO_CONSENSUS"
]

LABEL_GLOSS = {
    "TRUE": "TRUE — proposition holds on the truth spectrum",
    "FALSE": "FALSE — proposition fails on the truth spectrum",
    "INDETERMINATE": "INDETERMINATE — stable middle-region truth",
    "META_INDETERMINATE": "META-INDETERMINATE — incoherent, self-defeating, or category-confused",
    "N_A": "N/A — answer not currently possible",
    "NO_CONSENSUS": "NO CONSENSUS — raters split"
}

class TruthEvaluation(BaseModel):
    label: TruthLabel
    pd_degree: float = Field(ge=0, le=1)
    pd_modality: float = Field(ge=0, le=1)
    tau_delta: float = Field(ge=0, le=1)
    authority_loading: float = Field(ge=0, le=1)
    explanation: str

class ConsensusEvaluation(BaseModel):
    label: TruthLabel
    label_votes: Dict[str, int]
    unanimous: bool
    pd_degree: float
    pd_modality: float
    tau_delta: float
    authority_loading: float
    axis_spread: Dict[str, float]
    explanations: List[str]
    raters: List[str]
    failed_raters: List[str] = []

    @property
    def agreement_note(self) -> str:
        return f"Votes: {self.label_votes}; unanimous={self.unanimous}"
```
